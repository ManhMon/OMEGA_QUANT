+    if tp == ctypes.py_object:
+        return True

+    if tp == ctypes.c_void_p:

+def _resolve_backing_allocation(obj: object, ctypes_registry: dict) -> tuple[object, int, int]:
+    # Recursively trace a ctypes object back to its root allocation, base address, and byte offset.
+    # Returns a 3-tuple: (root_owner, root_base_address, offset)
+    curr = obj
+    offset = 0
+    visited = set()
+    while True:
+        curr_id = id(curr)
+        if curr_id in visited:
+            # Raise ValueError when a cycle is detected in the backing chain rather than silently breaking
+            raise ValueError("Fail-closed: cycle detected in backing allocation hierarchy")
+        visited.add(curr_id)
+
+        b_base = getattr(curr, "_b_base_", None)
+        if b_base is not None:
+            try:

+            except (TypeError, ValueError) as e:

+            else:

+

+            found = False

+                def recurse_find_owner(container, visited_con) -> tuple[object, int]:
+                    con_id = id(container)
+                    if con_id in visited_con:
+                        return None, 0
+                    visited_con.add(con_id)
+
+                    if isinstance(container, dict):
+                        for v in container.values():
+                            res = recurse_find_owner(v, visited_con)
+                            if res[0] is not None:
+                                return res
+                    elif isinstance(container, (list, tuple, set)):
+                        for item in container:
+                            res = recurse_find_owner(item, visited_con)
+                            if res[0] is not None:
+                                return res
+                    else:
+                        try:
+                            base_addr = ctypes.addressof(container)
+                            extent = ctypes.sizeof(container)
+                            if base_addr <= ptr_val < base_addr + extent:
+                                return container, ptr_val - base_addr
+                        except (TypeError, ValueError):
+                            pass
+                    return None, 0
+
+                matched_owner, matched_offset = recurse_find_owner(objs, set())
+                if matched_owner is not None:
+                    offset += matched_offset
+                    curr = matched_owner
+                    continue
+
+            # Fail-closed immediately for any pointer/c_void_p whose root allocation cannot be established in context
+            raise ValueError(f"Fail-closed: unverifiable ctypes pointer address {hex(ptr_val)} found in graph")
+
+        objs = getattr(curr, "_objects", None)
+        if objs is not None and not isinstance(objs, (dict, list, tuple)):
+            curr = objs
+            continue
+
+        break
+    try:
+        base_addr = ctypes.addressof(curr)
+    except (TypeError, ValueError):
+        base_addr = 0
+    return curr, base_addr, offset
+
+
+def _clone_recursive(obj: object, memo: dict, ctypes_memo: dict, in_progress: set, ctypes_registry: dict) -> object:
+    obj_id = id(obj)
+    if obj_id in memo:
+        return memo[obj_id]
+
+    if obj_id in in_progress:
+        raise ValueError("Fail-closed: cyclic reference through immutable container detected")
+
+    in_progress.add(obj_id)
+    try:
+        if isinstance(obj, np.ndarray):
+            if obj.dtype == 'O':
+                # Recursively deep clone each object cell inside object-dtype NumPy arrays
+                copied = np.empty_like(obj)
+                memo[obj_id] = copied
+                for idx in range(obj.size):
+                    copied.flat[idx] = _clone_recursive(obj.flat[idx], memo, ctypes_memo, in_progress, ctypes_registry)
+                return copied
+            else:
+                if obj.base is not None and isinstance(obj.base, np.ndarray):
+                    # Preserve geometry offsets for dependent ndarray views
+                    parent_offset = obj.ctypes.data - obj.base.ctypes.data
+                    base_cloned = _clone_recursive(obj.base, memo, ctypes_memo, in_progress, ctypes_registry)
+                    copied = np.ndarray(obj.shape, dtype=obj.dtype, buffer=base_cloned, offset=parent_offset, strides=obj.strides)
+                else:
+                    copied = obj.copy()
+                memo[obj_id] = copied
+                return copied
+
+        if isinstance(obj, bytearray):
+            copied = bytearray(obj)
+            memo[obj_id] = copied
+            return copied
+
+        if isinstance(obj, memoryview):
+            # Reject cloning any non-c-contiguous memoryview
+            if not obj.c_contiguous:
+                raise ValueError("Fail-closed: non-C-contiguous memoryview cloning not supported")
+
+            # Clone base
+            base_cloned = _clone_recursive(obj.obj, memo, ctypes_memo, in_progress, ctypes_registry)
+
+            # Force base to 1D bytes first so slicing is correct in byte coordinates
+            cloned_base_bytes = memoryview(base_cloned).cast('B')
+
+            try:
+                base_addr = ctypes.pythonapi.PyObject_GetBuffer  # type: ignore[attr-defined]
+                view_addr = ctypes.pythonapi.PyObject_GetBuffer  # type: ignore[attr-defined]
+                raise ValueError("Buffer address unresolved")
+            except Exception as e:
+                raise ValueError(f"Fail-closed: memoryview geometry address cannot be verified: {e}")
+
+        if isinstance(obj, list):
+            copied_list = []
+            memo[obj_id] = copied_list  # Register memo early for cycle protection
+            for item in obj:
+                copied_list.append(_clone_recursive(item, memo, ctypes_memo, in_progress, ctypes_registry))
+            return copied_list
+
+        if isinstance(obj, tuple):
+            cloned_elements = [_clone_recursive(item, memo, ctypes_memo, in_progress, ctypes_registry) for item in obj]
+            copied_tuple = tuple(cloned_elements)
+            memo[obj_id] = copied_tuple
+            return copied_tuple
+
+        if isinstance(obj, dict):
+            copied_dict = {}
+            memo[obj_id] = copied_dict  # Register memo early
+            for k, v in obj.items():
+                copied_dict[_clone_recursive(k, memo, ctypes_memo, in_progress, ctypes_registry)] = _clone_recursive(v, memo, ctypes_memo, in_progress, ctypes_registry)
+            return copied_dict
+
+        if isinstance(obj, set):
+            copied_set = set()
+            memo[obj_id] = copied_set  # Register memo early
+            for item in obj:
+                copied_set.add(_clone_recursive(item, memo, ctypes_memo, in_progress, ctypes_registry))
+            return copied_set
+
+        if isinstance(obj, frozenset):
+            copied_frozenset = frozenset(_clone_recursive(item, memo, ctypes_memo, in_progress, ctypes_registry) for item in obj)
+            memo[obj_id] = copied_frozenset
+            return copied_frozenset
+
+        if isinstance(obj, (int, float, str, bytes, type(None))):
+            return obj
+
+        # Explicit clone branch for ctypes._SimpleCData (scalars c_int, c_void_p, py_object...)
+        if isinstance(obj, ctypes._SimpleCData):
+            if isinstance(obj, ctypes.py_object):
+                val = obj.value
+                cloned_val = _clone_recursive(val, memo, ctypes_memo, in_progress, ctypes_registry)
+                copied = ctypes.py_object(cloned_val)
+            elif isinstance(obj, ctypes.c_void_p):
+                # Prevent c_void_p address leakage by safely re-routing to cloned root + offset from registry if found
+                addr_val = obj.value or 0
+                if addr_val == 0:
+                    copied = ctypes.c_void_p()
+                else:
+                    found = False
+                    for base_addr, (root_obj, extent) in ctypes_registry.items():
+                        if base_addr <= addr_val < base_addr + extent:
+                            offset = addr_val - base_addr
+                            cloned_root = _clone_recursive(root_obj, memo, ctypes_memo, in_progress, ctypes_registry)
+                            cloned_addr = ctypes.addressof(cloned_root) + offset
+                            copied = ctypes.c_void_p(cloned_addr)
+                            found = True
+                            break
+                    if not found:
+                        raise ValueError(f"Fail-closed: unresolved ctypes.c_void_p native address {hex(addr_val)}")
+            else:
+                copied = type(obj)(obj.value)
+            memo[obj_id] = copied
+            # SimpleCData does not own shared complex backing allocations, but memoize base address if any
+            try:
+                addr = ctypes.addressof(obj)
+                if addr > 0:
+                    ctypes_memo[(addr, 0, type(obj))] = copied
+            except (TypeError, ValueError):
+                pass
+            return copied
+
+        # Clone ctypes Array / Structure / Pointer and backing allocations
+        if isinstance(obj, (ctypes.Structure, ctypes.Union, ctypes.Array, ctypes._Pointer)):
+            try:
+                # Avoid collisions between pointer wrappers and pointee targets via memo key of base_address, offset, and type
+                root, base_addr, offset = _resolve_backing_allocation(obj, ctypes_registry)
+                memo_key = (base_addr, offset, type(obj))
+                if base_addr > 0 and memo_key in ctypes_memo:
+                    return ctypes_memo[memo_key]
+
+                if hasattr(obj, "_length_"):  # ctypes Array
+                    copied = type(obj)()
+                    memo[obj_id] = copied
+                    if base_addr > 0:
+                        ctypes_memo[memo_key] = copied
+                    # Early instantiation before cloning elements to protect array cycle
+                    for idx in range(len(obj)):
+                        copied[idx] = _clone_recursive(obj[idx], memo, ctypes_memo, in_progress, ctypes_registry)
+                    return copied
+                elif isinstance(obj, ctypes._Pointer):
+                    if not obj:  # NULL pointer
+                        copied = type(obj)()
+                        memo[obj_id] = copied
+                        if base_addr > 0:
+                            ctypes_memo[memo_key] = copied
+                        return copied
+                    else:
+                        # Absolutely forbid obj.contents dereference or cloning independent pointers not verified in registry
+                        if base_addr == 0:
+                            raise ValueError("Fail-closed: unverifiable non-NULL ctypes pointer found in graph")
+
+                        # Register placeholder pointer before clone to protect cyclic references
+                        copied = type(obj)()
+                        memo[obj_id] = copied
+                        ctypes_memo[memo_key] = copied
+
+                        # Reroute to cloned_root + matching offset
+                        cloned_root = _clone_recursive(root, memo, ctypes_memo, in_progress, ctypes_registry)
+                        cloned_ptr_val = ctypes.addressof(cloned_root) + offset
+
+                        # Write cloned_ptr_val safely into pointer memory without dereference
+                        ptr_address_var = ctypes.c_void_p(cloned_ptr_val)
+                        ctypes.memmove(ctypes.addressof(copied), ctypes.byref(ptr_address_var), ctypes.sizeof(copied))
+                        copied.__dict__["_lifetime_guard"] = cloned_root
+                        return copied
+                else:  # Structure / Union
+                    # Absolutely block fail-closed bypasses via nested unions containing pointer/py_object
+                    if _structure_or_union_contains_pointer_bearing_union(type(obj)):
+                        raise ValueError("Fail-closed: structure/union contains nested pointer-bearing Union")
+
+                    # Special safe handling for POD ctypes.Union:
+                    if isinstance(obj, ctypes.Union):
+                        copied = type(obj)()
+                        memo[obj_id] = copied
+                        if base_addr > 0:
+                            ctypes_memo[memo_key] = copied
+                        ctypes.memmove(ctypes.addressof(copied), ctypes.addressof(obj), ctypes.sizeof(obj))
+                        return copied
+
+                    # Normal Structure handling
+                    copied = type(obj)()
+                    memo[obj_id] = copied  # Register memo early
+                    if base_addr > 0:
+                        ctypes_memo[memo_key] = copied
+
+                    # Only raw memmove for purely POD recursive Structures (no pointers or py_object)
+                    has_ptr_or_py_obj = _contains_pointer_or_py_object_recursive(type(obj))
+                    if not has_ptr_or_py_obj:
+                        ctypes.memmove(ctypes.addressof(copied), ctypes.addressof(obj), ctypes.sizeof(obj))
+
+                        # Walk inline arrays / nested Structures for deep clone (no pointer/py_object)
+                        for field_info in getattr(type(obj), "_fields_", []):
+                            if len(field_info) == 3:
+                                continue
+                            field_name = field_info[0]
+                            field_type = field_info[1]
+                            if issubclass(field_type, ctypes.Array):
+                                dest_arr = getattr(copied, field_name)
+                                src_arr = getattr(obj, field_name)
+                                for idx in range(len(dest_arr)):
+                                    dest_arr[idx] = _clone_recursive(src_arr[idx], memo, ctypes_memo, in_progress, ctypes_registry)
+                            elif issubclass(field_type, ctypes.Structure):
+                                val_field = getattr(obj, field_name)
+                                cloned_sub = _clone_recursive(val_field, memo, ctypes_memo, in_progress, ctypes_registry)
+                                setattr(copied, field_name, cloned_sub)
+                    else:
+                        # For Structures containing py_object or pointers: assign field-by-field in a type-aware way
+                        # Prevent incorrect decref or memory corruption absolutely
+                        for field_info in getattr(type(obj), "_fields_", []):
+                            field_name = field_info[0]
+                            field_type = field_info[1]
+                            if len(field_info) == 3:  # Bitfields
+                                val_field = getattr(obj, field_name)
+                                setattr(copied, field_name, val_field)
+                                continue
+
+                            if field_type == ctypes.py_object:
+                                val_field = getattr(obj, field_name)
+                                cloned_py_obj = _clone_recursive(val_field, memo, ctypes_memo, in_progress, ctypes_registry)
+                                setattr(copied, field_name, cloned_py_obj)
+                            elif field_type == ctypes.c_void_p:
+                                # Type-aware c_void_p field cloning: registry reroute or fail-closed
+                                addr_val = getattr(obj, field_name) or 0
+                                if addr_val == 0:
+                                    setattr(copied, field_name, 0)
+                                else:
+                                    found = False
+                                    for base_addr, (root_obj, extent) in ctypes_registry.items():
+                                        if base_addr <= addr_val < base_addr + extent:
+                                            offset = addr_val - base_addr
+                                            cloned_root = _clone_recursive(root_obj, memo, ctypes_memo, in_progress, ctypes_registry)
+                                            cloned_addr = ctypes.addressof(cloned_root) + offset
+                                            setattr(copied, field_name, cloned_addr)
+                                            copied.__dict__[f"_lifetime_guard_{field_name}"] = cloned_root
+                                            found = True
+                                            break
+                                    if not found:
+                                        raise ValueError(f"Fail-closed: unresolved nested c_void_p field '{field_name}' address {hex(addr_val)}")
+                            else:
+                                val_field = getattr(obj, field_name)
+                                cloned_val = _clone_recursive(val_field, memo, ctypes_memo, in_progress, ctypes_registry)
+                                setattr(copied, field_name, cloned_val)
+                                if issubclass(field_type, ctypes._Pointer):
+                                    copied.__dict__[f"_lifetime_guard_{field_name}"] = cloned_val
+                    return copied
+            except Exception as e:
+                if isinstance(e, ValueError) and "Fail-closed" in str(e):
+                    raise e
+                raise ValueError(f"ctypes clone failed for type {type(obj)}") from e
+
+        try:
+            cls = type(obj)
+            copied = cls.__new__(cls)
+            memo[obj_id] = copied  # Register memo early
+            if hasattr(obj, "__dict__"):
+                copied.__dict__.update(_clone_recursive(obj.__dict__, memo, ctypes_memo, in_progress, ctypes_registry))
+            for base in cls.__mro__:
+                slots = getattr(base, "__slots__", None)
+                if slots is not None:
+                    if isinstance(slots, str):
+                        slots = [slots]
+                    for slot in slots:
+                        if slot in ("__dict__", "__weakref__"):
+                            continue
+                        slot_name = slot
+                        if slot.startswith("__") and not slot.endswith("__"):
+                            slot_name = f"_{base.__name__}{slot}"
+                        if hasattr(obj, slot_name):
+                            setattr(copied, slot_name, _clone_recursive(getattr(obj, slot_name), memo, ctypes_memo, in_progress, ctypes_registry))
+            return copied
+        except Exception as e:
+            raise ValueError(f"Cannot clone object of type {type(obj)}") from e
+    finally:
+        in_progress.remove(obj_id)
